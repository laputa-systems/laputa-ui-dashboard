# @joshuarli/ui

Compact, editorial design system primitives for Solid.

## Installation

```sh
pnpm add @joshuarli/ui
```

## Usage

```ts
import "@joshuarli/ui/styles.css";

import {
  Badge,
  Button,
  CollapsiblePanel,
  EmptyState,
  ErrorState,
  Icon,
  Meter,
  Modal,
  Notice,
  PageHeader,
  Pagination,
  Panel,
  Segmented,
  Skeleton
} from "@joshuarli/ui";
```
